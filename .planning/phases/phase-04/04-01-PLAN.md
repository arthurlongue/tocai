  <action>
    Extend src/engine/AudioEngine.ts:
    
    1. Add private property: baseBpm: number = 120
    
    2. Add method:
       ```typescript
       public setSpeed(multiplier: number): void {
         const newBpm = this.baseBpm * multiplier;
         Tone.getTransport().bpm.value = newBpm;
       }
       ```
    
    3. Add method to set base BPM from song:
       ```typescript
       public setBaseBpm(bpm: number): void {
         this.baseBpm = bpm;
       }
       ```
    
    Create src/hooks/useSpeedControl.ts:
    
    1. Imports: useCallback, useState, AudioEngine, Tone
    
    2. Hook implementation:
       ```typescript
       export const useSpeedControl = () => {
         const engine = AudioEngine.getInstance();
         const [speed, setSpeedState] = useState(1);
         
         const setSpeed = useCallback((multiplier: number) => {
           const clamped = Math.max(0.25, Math.min(2.0, multiplier));
           setSpeedState(clamped);
           engine.setSpeed(clamped);
         }, [engine]);
         
         const resetSpeed = useCallback(() => {
           setSpeedState(1);
           engine.setSpeed(1);
         }, [engine]);
         
         return { speed, setSpeed, resetSpeed };
       };
       ```

    - Test: setSpeed clamps to 0.25 minimum
    - Test: setSpeed clamps to 2.0 maximum
    - Test: resetSpeed returns to 1.0
